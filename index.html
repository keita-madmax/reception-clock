  <script>

    // 1. YouTube 動画のIDを指定（URLの "v=" の後ろの英数字）

    // 例: https://www.youtube.com/watch?v=dQw4w9WgXcQ なら "dQw4w9WgXcQ"

    const YOUTUBE_VIDEO_ID = '-2o45BFOddQ&t=1988s'; // ※4K自然風景の例


    // 2. YouTube IFrame APIの読み込み

    const tag = document.createElement('script');

    tag.src = "https://www.youtube.com/iframe_api";

    const firstScriptTag = document.getElementsByTagName('script')[0];

    firstScriptTag.parentNode.insertBefore(tag, firstScriptTag);


    let player;

    function onYouTubeIframeAPIReady() {

      player = new YT.Player('player', {

        videoId: YOUTUBE_VIDEO_ID,

        playerVars: {

          autoplay: 1,

          controls: 0,        // シークバーや設定ボタンを非表示

          showinfo: 0,

          modestbranding: 1,

          loop: 1,

          playlist: YOUTUBE_VIDEO_ID, // ループ再生に必須

          fs: 0,

          cc_load_policy: 0,

          iv_load_policy: 3,

          autohide: 1,

          mute: 1             // 自動再生のためにミュート必須

        },

        events: {

          onReady: (e) => e.target.playVideo()

        }

      });

    }


    // 3. 時計の更新処理

    function updateClock() {

      const now = new Date();

      

      // 日付フォーマット (例: 10月24日 金曜日)

      const options = { month: 'numeric', day: 'numeric', weekday: 'short' };

      document.getElementById('date-display').textContent = now.toLocaleDateString('ja-JP', options);


      // 時刻フォーマット (例: 14:05)

      const hours = String(now.getHours()).padStart(2, '0');

      const minutes = String(now.getMinutes()).padStart(2, '0');

      document.getElementById('time-display').textContent = `${hours}:${minutes}`;

    }

    setInterval(updateClock, 1000);

    updateClock();
